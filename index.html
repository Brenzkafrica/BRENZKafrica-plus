const match = item.match(regex); return match ? decodeXML( match[1] ).trim() : ""; } function getLink(item) { const link = getTag( item, "link" ); if (link) { return link; } const guid = getTag( item, "guid" ); return true; } ); allNews.sort( function(a, b) { const dateA = a.date ? new Date( a.date ).getTime() : 0; const dateB = b.date ? new Date( b.date ).getTime() : 0; return dateB - dateA; } ); const news = allNews.slice( 0, 20 );
